```swift
import Foundation

public struct GitError: Error, Equatable, CustomStringConvertible {
    public let message: String
    public init(_ message: String) { self.message = message }
    public var description: String { message }
}

/// Runs the git CLI in one repo. The only place diffary spawns git.
public struct Git: Sendable {
    public let repo: URL
    public init(repo: URL) { self.repo = repo }

    /// Returns stdout. Throws GitError carrying stderr's first line when the
    /// exit code isn't in `allowExitCodes`.
    public func run(_ args: [String], allowExitCodes: Set<Int32> = [0]) throws -> String {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: "/usr/bin/env")
        process.arguments = ["git", "-C", repo.path] + args
        let out = Pipe(), err = Pipe()
        process.standardOutput = out
        process.standardError = err
        try process.run()

        // Drain stderr concurrently so a chatty stderr can't block stdout.
        var errData = Data()
        let group = DispatchGroup()
        group.enter()
        DispatchQueue.global().async {
            errData = err.fileHandleForReading.readDataToEndOfFile()
            group.leave()
        }
        let outData = out.fileHandleForReading.readDataToEndOfFile()
        group.wait()
        process.waitUntilExit()

        guard allowExitCodes.contains(process.terminationStatus) else {
            let first = String(decoding: errData, as: UTF8.self)
                .split(separator: "\n").first.map(String.init)
            throw GitError(first ?? "git \(args.first ?? "") exited \(process.terminationStatus)")
        }
        return String(decoding: outData, as: UTF8.self)
    }
}
```

```swift
        _ = try git.run(["config", "user.email", "test@example.com"])
        _ = try git.run(["config", "user.name", "test"])

        let file = url.appendingPathComponent(path)
        try FileManager.default.createDirectory(
            at: file.deletingLastPathComponent(), withIntermediateDirectories: true)
        try text.write(to: file, atomically: true, encoding: .utf8)

    func writeData(_ path: String, _ data: Data) throws {
        try data.write(to: url.appendingPathComponent(path))
    }

    func run(_ args: String...) throws -> String { try git.run(args) }

    func commitAll(_ message: String = "c") throws {
        try run("add", "-A")
        try run("commit", "-q", "-m", message)
    }
```

`Sources/DiffaryCore/GitChanges.swift`:
```swift
import Foundation

public enum ChangeStatus: String, Sendable {
    case added = "A", modified = "M", deleted = "D", renamed = "R", untracked = "?"
}

public struct FileChange: Sendable, Equatable, Identifiable {
    public var id: String { path }
    public let path: String
    public let oldPath: String?
    public let status: ChangeStatus
    /// nil for binary files.
    public let additions: Int?
    public let deletions: Int?
}

public struct ChangeSet: Sendable, Equatable {
    /// merge-base commit sha the files are diffed against.
    public let base: String
    public let files: [FileChange]
}

/// Lists what the working tree changed vs merge-base(baseBranch, HEAD).
public struct GitChanges: Sendable {
    public let git: Git
    public let baseBranch: String

    public init(repo: URL, baseBranch: String) {
        self.git = Git(repo: repo)
        self.baseBranch = baseBranch
    }

    public func load() throws -> ChangeSet {
        _ = try git.run(["rev-parse", "--is-inside-work-tree"])
        let base = try mergeBase()
        let numstat = parseNumstat(try git.run(["diff", "--numstat", "-M", "-z", base]))
        var files = parseNameStatus(try git.run(["diff", "--name-status", "-M", "-z", base]))
            .map { entry in
                let counts = numstat[entry.path]
                return FileChange(path: entry.path, oldPath: entry.oldPath, status: entry.status,
                                  additions: counts?.0, deletions: counts?.1)
            }
        let untracked = try git.run(["ls-files", "--others", "--exclude-standard", "-z"])
            .split(separator: "\0").map(String.init)
        files += untracked.map { path in
            FileChange(path: path, oldPath: nil, status: .untracked,
                       additions: lineCount(git.repo.appendingPathComponent(path)), deletions: 0)
        }
        return ChangeSet(base: base, files: files.sorted { $0.path < $1.path })
    }

    func mergeBase() throws -> String {
        let candidates = baseBranch == "master" ? ["master", "main"] : [baseBranch]
        for branch in candidates {
            if (try? git.run(["rev-parse", "--verify", "--quiet", "\(branch)^{commit}"])) != nil {
                return try git.run(["merge-base", branch, "HEAD"])
                    .trimmingCharacters(in: .whitespacesAndNewlines)
            }
        }
        throw GitError("base branch '\(baseBranch)' not found")
    }

    /// `-z` name-status: "M\0path\0", renames "R100\0old\0new\0".
    func parseNameStatus(_ raw: String) -> [(path: String, oldPath: String?, status: ChangeStatus)] {
        var fields = raw.split(separator: "\0", omittingEmptySubsequences: false).map(String.init)[...]
        var result: [(String, String?, ChangeStatus)] = []
        while let code = fields.popFirst(), !code.isEmpty {
            switch code.first {
            case "R", "C":
                guard let old = fields.popFirst(), let new = fields.popFirst() else { break }
                result.append((new, old, .renamed))
            case "A": if let p = fields.popFirst() { result.append((p, nil, .added)) }
            case "D": if let p = fields.popFirst() { result.append((p, nil, .deleted)) }
            default: if let p = fields.popFirst() { result.append((p, nil, .modified)) }
            }
        }
        return result
    }

    /// `-z` numstat: "add\tdel\tpath\0", renames "add\tdel\t\0old\0new\0". Binary shows "-".
    func parseNumstat(_ raw: String) -> [String: (Int?, Int?)] {
        var fields = raw.split(separator: "\0", omittingEmptySubsequences: false).map(String.init)[...]
        var result: [String: (Int?, Int?)] = [:]
        while let head = fields.popFirst(), !head.isEmpty {
            let parts = head.split(separator: "\t", maxSplits: 2, omittingEmptySubsequences: false)
            guard parts.count == 3 else { continue }
            var path = String(parts[2])
            if path.isEmpty { _ = fields.popFirst(); path = fields.popFirst() ?? "" }
            result[path] = (Int(parts[0]), Int(parts[1]))
        }
        return result
    }

    /// Lines in an untracked file; nil if it looks binary (contains NUL).
    func lineCount(_ url: URL) -> Int? {
        guard let data = try? Data(contentsOf: url) else { return nil }
        if data.contains(0) { return nil }
        let newlines = data.reduce(0) { $1 == 0x0A ? $0 + 1 : $0 }
        return newlines + (data.last.map { $0 == 0x0A ? 0 : 1 } ?? 0)
    }
}
```

`Sources/DiffaryCore/DiffParser.swift`:
```swift
import Foundation

public enum DiffLineKind: Sendable, Equatable { case context, added, removed }

public struct DiffLine: Sendable, Equatable {
    public let kind: DiffLineKind
    public let text: String
    public let oldNumber: Int?
    public let newNumber: Int?
}

public struct SideBySideRow: Sendable, Equatable {
    public let left: DiffLine?
    public let right: DiffLine?
}

public struct Hunk: Sendable, Equatable {
    public let header: String
    public let lines: [DiffLine]

    /// Context on both sides; each run of removals is paired line-by-line with
    /// the run of additions that follows it.
    public var sideBySideRows: [SideBySideRow] {
        var rows: [SideBySideRow] = []
        var removed: [DiffLine] = [], added: [DiffLine] = []
        func flush() {
            for i in 0..<max(removed.count, added.count) {
                rows.append(SideBySideRow(left: i < removed.count ? removed[i] : nil,
                                          right: i < added.count ? added[i] : nil))
            }
            removed = []; added = []
        }
        for line in lines {
            switch line.kind {
            case .removed:
                if !added.isEmpty { flush() }
                removed.append(line)
            case .added: added.append(line)
            case .context: flush(); rows.append(SideBySideRow(left: line, right: line))
            }
        }
        flush()
        return rows
    }
}

public enum FileDiff: Sendable, Equatable {
    case text([Hunk]), binary, tooLarge(lines: Int), empty
}

public enum DiffParser {
    public static let maxLines = 20_000

    public static func parse(_ raw: String) -> FileDiff {
        var lines = raw.components(separatedBy: "\n")
        if lines.last == "" { lines.removeLast() }
        if lines.contains(where: { $0.hasPrefix("Binary files ") }) { return .binary }
        if lines.count > maxLines { return .tooLarge(lines: lines.count) }

        var hunks: [Hunk] = []
        var header: String?
        var body: [DiffLine] = []
        var old = 0, new = 0

        func close() {
            if let header { hunks.append(Hunk(header: header, lines: body)) }
            body = []
        }

        for line in lines {
            if line.hasPrefix("@@") {
                close()
                header = line
                (old, new) = startNumbers(line)
                continue
            }
            guard header != nil, let first = line.first else { continue }
            let text = String(line.dropFirst())
            switch first {
            case " ": body.append(DiffLine(kind: .context, text: text, oldNumber: old, newNumber: new)); old += 1; new += 1
            case "-": body.append(DiffLine(kind: .removed, text: text, oldNumber: old, newNumber: nil)); old += 1
            case "+": body.append(DiffLine(kind: .added, text: text, oldNumber: nil, newNumber: new)); new += 1
            default: continue  // "\ No newline at end of file"
            }
        }
        close()
        return hunks.isEmpty ? .empty : .text(hunks)
    }

    /// "@@ -12,3 +14,5 @@" → (12, 14)
    static func startNumbers(_ header: String) -> (Int, Int) {
        let parts = header.split(separator: " ")
        func start(_ prefix: Character) -> Int {
            guard let p = parts.first(where: { $0.first == prefix }) else { return 1 }
            return Int(p.dropFirst().split(separator: ",").first ?? "1") ?? 1
        }
        return (start("-"), start("+"))
    }
}
```

```swift
import Foundation

public struct DiffOptions: Sendable, Equatable {
    public var contextLines: Int
    public var ignoreWhitespace: Bool
    public init(contextLines: Int, ignoreWhitespace: Bool) {
        self.contextLines = contextLines
        self.ignoreWhitespace = ignoreWhitespace
    }
}

/// Loads the diff for one file vs the change set's base.
public struct DiffLoader: Sendable {
    public let git: Git
    public init(repo: URL) { git = Git(repo: repo) }

    public func load(_ change: FileChange, base: String, options: DiffOptions) throws -> FileDiff {
        var args = ["diff", "--no-color", "--no-ext-diff", "-U\(options.contextLines)"]
        if options.ignoreWhitespace { args.append("-w") }
        if change.status == .untracked {
            // --no-index exits 1 when the files differ.
            args += ["--no-index", "--", "/dev/null", change.path]
            return DiffParser.parse(try git.run(args, allowExitCodes: [0, 1]))
        }
        args += ["-M", base, "--"] + [change.oldPath, change.path].compactMap { $0 }
        return DiffParser.parse(try git.run(args))
    }
}
```

```swift
    public var syntaxHighlight = true
    public var font = "JetBrains Mono"
    public var fontSize = 13.0

    public var diffOptions: DiffOptions {

    }

public struct SettingsLoad: Sendable, Equatable {
    public let settings: Settings
    public let warnings: [String]
}

public enum SettingsLoader {
    public static func configDir(env: [String: String] = ProcessInfo.processInfo.environment) -> URL {
        let xdg = env["XDG_CONFIG_HOME"].flatMap { $0.isEmpty ? nil : $0 }
        let root = xdg ?? ((env["HOME"] ?? NSHomeDirectory()) + "/.config")
        return URL(fileURLWithPath: root).appendingPathComponent("diffary")
    }

    public static func defaultConfigFile(env: [String: String] = ProcessInfo.processInfo.environment) -> URL {
        configDir(env: env).appendingPathComponent("config.toml")
    }

    /// File first, then overrides. Never throws: problems become warnings and
    /// the affected key keeps its default.
    public static func load(file: URL, overrides: [String: String]) -> SettingsLoad {
        var settings = Settings()
        var warnings: [String] = []
        var values: [(String, String)] = []

        if let text = try? String(contentsOf: file, encoding: .utf8) {
            do {
                values = flatten(try TOMLTable(string: text), prefix: "")
            } catch {
                warnings.append("\(file.lastPathComponent): \(error)")
            }
        }
        values += overrides.sorted { $0.key < $1.key }

        for (key, value) in values {
            guard let apply = setters[key] else {
                warnings.append("unknown config key '\(key)'")
                continue
            }
            if !apply(&settings, value) {
                warnings.append("invalid value '\(value)' for '\(key)', using default")
            }
        }
        return SettingsLoad(settings: settings, warnings: warnings)
    }

    /// Nested tables → dotted keys, scalar values → strings.
    static func flatten(_ table: TOMLTable, prefix: String) -> [(String, String)] {
        var out: [(String, String)] = []
        for (key, value) in table {
            let full = prefix.isEmpty ? key : "\(prefix).\(key)"
            if let sub = value.table { out += flatten(sub, prefix: full) }
            else if let s = value.string { out.append((full, s)) }
            else if let i = value.int { out.append((full, String(i))) }
            else if let d = value.double { out.append((full, String(d))) }
            else if let b = value.bool { out.append((full, String(b))) }
            else { out.append((full, "\(value)")) }
        }
        return out
    }

    typealias Setter = @Sendable (inout Settings, String) -> Bool

    static let setters: [String: Setter] = [
        "base_branch": { s, v in guard !v.isEmpty else { return false }; s.baseBranch = v; return true },
        "layout.sidebar": { s, v in set(&s.sidebar, SidebarPosition(rawValue: v)) },
        "layout.sidebar_size": { s, v in set(&s.sidebarSize, Double(v).flatMap { (0.1...0.9).contains($0) ? $0 : nil }) },
        "layout.diff_position": { s, v in set(&s.diffPosition, DiffPosition(rawValue: v)) },
        "diff.view": { s, v in set(&s.diffView, DiffViewMode(rawValue: v)) },
        "diff.context_lines": { s, v in set(&s.contextLines, Int(v).flatMap { (0...1000).contains($0) ? $0 : nil }) },
        "diff.ignore_whitespace": { s, v in set(&s.ignoreWhitespace, Bool(v)) },
        "diff.syntax_highlight": { s, v in set(&s.syntaxHighlight, Bool(v)) },
        "appearance.font": { s, v in guard !v.isEmpty else { return false }; s.font = v; return true },
        "appearance.font_size": { s, v in set(&s.fontSize, Double(v).flatMap { (6...72).contains($0) ? $0 : nil }) },
        "appearance.theme": { s, v in guard !v.isEmpty else { return false }; s.theme = v; return true },
    ]

    private static func set<T>(_ target: inout T, _ value: T?) -> Bool {
        guard let value else { return false }
        target = value
        return true
    }
}
```

```swift
import Foundation
import TOMLKit

public struct RGBA: Sendable, Equatable {
    public var r, g, b, a: Double
    public init(r: Double, g: Double, b: Double, a: Double) { self.r = r; self.g = g; self.b = b; self.a = a }

    /// "#rrggbb" or "#rrggbbaa".
    public init?(hex: String) {
        guard hex.hasPrefix("#") else { return nil }
        let digits = hex.dropFirst()
        guard digits.count == 6 || digits.count == 8, let v = UInt32(digits, radix: 16) else { return nil }
        let full = digits.count == 6 ? (v << 8) | 0xFF : v
        r = Double((full >> 24) & 0xFF) / 255
        g = Double((full >> 16) & 0xFF) / 255
        b = Double((full >> 8) & 0xFF) / 255
        a = Double(full & 0xFF) / 255
    }

    public var hex: String {
        String(format: "#%02x%02x%02x%02x",
               Int((r * 255).rounded()), Int((g * 255).rounded()),
               Int((b * 255).rounded()), Int((a * 255).rounded()))
    }
}

    public var isDark: Bool
    public var background, foreground, sidebarBackground, selection: RGBA
    public var addedBg, addedFg, removedBg, removedFg: RGBA

        "removed_bg": \.removedBg, "removed_fg": \.removedFg,
        "hunk_header": \.hunkHeader, "line_number": \.lineNumber,

        addedBg: c("#1a7f3720"), addedFg: c("#1a7f37"), removedBg: c("#cf222e20"), removedFg: c("#cf222e"),

        statusAdded: c("#1a7f37"), statusModified: c("#9a6700"), statusDeleted: c("#cf222e"), statusRenamed: c("#0969da"),
        syntax: ["keyword": c("#cf222e"), "string": c("#0a3069"), "comment": c("#6e7781"),

public enum ThemeLoader {
    public static func resolve(name: String, systemIsDark: Bool, themesDir: URL) -> (theme: Theme, warnings: [String]) {
        let auto = systemIsDark ? Theme.dark : Theme.light
        switch name {
        case "auto": return (auto, [])
        case "dark": return (.dark, [])
        case "light": return (.light, [])
        default: break
        }
        let file = themesDir.appendingPathComponent("\(name).toml")
        guard let text = try? String(contentsOf: file, encoding: .utf8) else {
            return (auto, ["theme '\(name)' not found at \(file.path), using auto"])
        }
        let table: TOMLTable
        do { table = try TOMLTable(string: text) } catch {
            return (auto, ["theme '\(name)': \(error)"])
        }

        var warnings: [String] = []
        var theme = table["base"]?.string == "light" ? Theme.light : Theme.dark
        for (key, value) in table {
            if key == "base" { continue }
            if key == "syntax", let syntax = value.table {
                for (sk, sv) in syntax {
                    guard Theme.syntaxKeys.contains(sk), let color = sv.string.flatMap(RGBA.init(hex:)) else {
                        warnings.append("theme '\(name)': bad syntax.\(sk)"); continue
                    }
                    theme.syntax[sk] = color
                }
                continue
            }
            guard let path = Theme.colorKeys[key] else {
                warnings.append("theme '\(name)': unknown key '\(key)'"); continue
            }
            guard let color = value.string.flatMap(RGBA.init(hex:)) else {
                warnings.append("theme '\(name)': bad color for '\(key)'"); continue
            }
            theme[keyPath: path] = color
        }
        return (theme, warnings)
    }
}
```

```swift
import ArgumentParser
import Foundation

public struct Launch: Sendable, Equatable {
    public let repo: URL
    public let configFile: URL
    public let overrides: [String: String]

    public static func parse(_ args: [String], cwd: URL, env: [String: String]) throws -> Launch {
        try LaunchCommand.parse(args).resolve(cwd: cwd, env: env)
    }
}

public struct LaunchCommand: ParsableCommand {
    public static let configuration = CommandConfiguration(
        commandName: "diffary",
        abstract: "Ghostty terminal + live branch diff.",
        version: "0.1.0")

    @Argument(help: "Repo to watch. Default: current directory.")
    var repo: String?

    @Option(help: "Config file. Default: $XDG_CONFIG_HOME/diffary/config.toml.")

            repo: repo.map(abs) ?? cwd.standardizedFileURL,
            configFile: config.map(abs) ?? SettingsLoader.defaultConfigFile(env: env),
```

```swift
import CoreServices
import Foundation

/// FSEvents watcher with a trailing debounce. All state lives on `queue`.
public final class FileWatcher: @unchecked Sendable {
    private let paths: [URL]
    private let ignoring: [String]
    private let debounce: TimeInterval
    private let onChange: @Sendable () -> Void
    private let queue = DispatchQueue(label: "diffary.filewatcher")
    private var stream: FSEventStreamRef?
    private var pending: DispatchWorkItem?

    public init(paths: [URL], ignoring: [String] = [], debounce: TimeInterval = 0.3,
                onChange: @escaping @Sendable () -> Void) {
        self.paths = paths
        self.ignoring = ignoring
        self.debounce = debounce
        self.onChange = onChange
    }

    deinit { stop() }

    public func start() {
        var context = FSEventStreamContext(
            version: 0, info: Unmanaged.passUnretained(self).toOpaque(),
            retain: nil, release: nil, copyDescription: nil)
        let callback: FSEventStreamCallback = { _, info, count, eventPaths, _, _ in
            guard let info else { return }
            let watcher = Unmanaged<FileWatcher>.fromOpaque(info).takeUnretainedValue()
            let paths = unsafeBitCast(eventPaths, to: NSArray.self) as? [String] ?? []
            watcher.handle(Array(paths.prefix(count)))
        }
        let flags = UInt32(kFSEventStreamCreateFlagUseCFTypes | kFSEventStreamCreateFlagFileEvents
                           | kFSEventStreamCreateFlagNoDefer)
        guard let stream = FSEventStreamCreate(
            nil, callback, &context, paths.map(\.path) as CFArray,
            FSEventStreamEventId(kFSEventStreamEventIdSinceNow), 0.1, flags)
        else { return }
        self.stream = stream
        FSEventStreamSetDispatchQueue(stream, queue)
        FSEventStreamStart(stream)
    }

    public func stop() {
        guard let stream else { return }
        FSEventStreamStop(stream)
        FSEventStreamInvalidate(stream)
        FSEventStreamRelease(stream)
        self.stream = nil
        pending?.cancel()
    }

    private func handle(_ eventPaths: [String]) {
        let relevant = eventPaths.contains { path in !ignoring.contains { path.contains($0) } }
        guard relevant else { return }
        pending?.cancel()
        let work = DispatchWorkItem { [onChange] in onChange() }
        pending = work
        queue.asyncAfter(deadline: .now() + debounce, execute: work)
    }
}
```

```swift
        // own flags aren't parsed as Ghostty CLI actions.

        guard ghostty_init(1, &argv) == GHOSTTY_SUCCESS else { error = "ghostty_init failed"; return }

            },

            confirm_read_clipboard_cb: { userdata, confirm, state, _ in

            },
            write_clipboard_cb: { _, _, contents, count, _ in

                guard let userdata else { return }
```

```swift
import GhosttyKit

/// NSView hosting one Ghostty surface. Knows nothing about git.
final class TerminalView: NSView {
    private(set) var surface: ghostty_surface_t?
    private(set) var startupError: String?

    init(workingDirectory: URL) {
        super.init(frame: NSRect(x: 0, y: 0, width: 800, height: 600))
        guard let app = GhosttyRuntime.shared.app else {
            startupError = GhosttyRuntime.shared.error ?? "Ghostty failed to start"
            return
        }
        var cfg = ghostty_surface_config_new()
        cfg.platform_tag = GHOSTTY_PLATFORM_MACOS
        cfg.platform = ghostty_platform_u(macos: ghostty_platform_macos_s(
            nsview: Unmanaged.passUnretained(self).toOpaque()))
        cfg.userdata = Unmanaged.passUnretained(self).toOpaque()
        cfg.scale_factor = NSScreen.main?.backingScaleFactor ?? 2
        cfg.context = GHOSTTY_SURFACE_CONTEXT_WINDOW
        surface = workingDirectory.path.withCString { dir in
            cfg.working_directory = dir
            return ghostty_surface_new(app, &cfg)
        }
        if surface == nil { startupError = "ghostty_surface_new failed" }
    }

    required init?(coder: NSCoder) { fatalError("not used") }

    deinit { if let surface { ghostty_surface_free(surface) } }

    override var acceptsFirstResponder: Bool { true }

    override func becomeFirstResponder() -> Bool {
        if let surface { ghostty_surface_set_focus(surface, true) }
        return true
    }

    override func resignFirstResponder() -> Bool {
        if let surface { ghostty_surface_set_focus(surface, false) }
        return true
    }

    // MARK: Size

    override func setFrameSize(_ newSize: NSSize) {
        super.setFrameSize(newSize)
        updateSize()
    }

    override func viewDidChangeBackingProperties() {
        super.viewDidChangeBackingProperties()
        guard let surface, let window else { return }
        let scale = window.backingScaleFactor
        ghostty_surface_set_content_scale(surface, scale, scale)
        updateSize()
    }

    private func updateSize() {
        guard let surface else { return }
        let px = convertToBacking(bounds.size)
        ghostty_surface_set_size(surface, UInt32(max(px.width, 1)), UInt32(max(px.height, 1)))
    }

    // MARK: Keyboard

    override func keyDown(with event: NSEvent) { sendKey(event, event.isARepeat ? GHOSTTY_ACTION_REPEAT : GHOSTTY_ACTION_PRESS) }
    override func keyUp(with event: NSEvent) { sendKey(event, GHOSTTY_ACTION_RELEASE) }

    override func flagsChanged(with event: NSEvent) {
        guard let surface else { return }
        var key = keyEvent(event, GHOSTTY_ACTION_PRESS)
        key.text = nil
        _ = ghostty_surface_key(surface, key)
    }

    /// Let Ghostty see Cmd-shortcuts (copy/paste/etc. come from its keybinds).
    override func performKeyEquivalent(with event: NSEvent) -> Bool {
        guard window?.firstResponder === self, event.modifierFlags.contains(.command) else { return false }
        keyDown(with: event)
        return true
    }

    // ponytail: no IME/dead-key composition (interpretKeyEvents path). Port
    // keyDown + NSTextInputClient from SurfaceView_AppKit.swift if needed.
    private func sendKey(_ event: NSEvent, _ action: ghostty_input_action_e) {
        guard let surface else { return }
        var key = keyEvent(event, action)
        let text = action == GHOSTTY_ACTION_RELEASE ? nil : Self.ghosttyText(event)
        if let text {
            text.withCString { ptr in
                key.text = ptr
                _ = ghostty_surface_key(surface, key)
            }
        } else {
            _ = ghostty_surface_key(surface, key)
        }
    }

    private func keyEvent(_ event: NSEvent, _ action: ghostty_input_action_e) -> ghostty_input_key_s {
        var key = ghostty_input_key_s()
        key.action = action
        key.keycode = UInt32(event.keyCode)
        key.mods = Self.mods(event.modifierFlags)
        key.consumed_mods = Self.mods(event.modifierFlags.subtracting([.control, .command]))
        key.composing = false
        if event.type == .keyDown || event.type == .keyUp,
           let scalar = event.characters(byApplyingModifiers: [])?.unicodeScalars.first {
            key.unshifted_codepoint = scalar.value
        }
        return key
    }

    /// Same rules as Ghostty's NSEvent.ghosttyCharacters.
    private static func ghosttyText(_ event: NSEvent) -> String? {
        guard let chars = event.characters else { return nil }
        if chars.count == 1, let scalar = chars.unicodeScalars.first {
            if scalar.value < 0x20 { return event.characters(byApplyingModifiers: event.modifierFlags.subtracting(.control)) }
            if (0xF700...0xF8FF).contains(scalar.value) { return nil }
        }
        return chars
    }

    static func mods(_ flags: NSEvent.ModifierFlags) -> ghostty_input_mods_e {
        var raw = GHOSTTY_MODS_NONE.rawValue
        if flags.contains(.shift) { raw |= GHOSTTY_MODS_SHIFT.rawValue }
        if flags.contains(.control) { raw |= GHOSTTY_MODS_CTRL.rawValue }
        if flags.contains(.option) { raw |= GHOSTTY_MODS_ALT.rawValue }
        if flags.contains(.command) { raw |= GHOSTTY_MODS_SUPER.rawValue }
        if flags.contains(.capsLock) { raw |= GHOSTTY_MODS_CAPS.rawValue }
        return ghostty_input_mods_e(rawValue: raw)
    }

    // MARK: Mouse

    override func mouseDown(with e: NSEvent) { window?.makeFirstResponder(self); button(e, GHOSTTY_MOUSE_PRESS, GHOSTTY_MOUSE_LEFT) }
    override func mouseUp(with e: NSEvent) { button(e, GHOSTTY_MOUSE_RELEASE, GHOSTTY_MOUSE_LEFT) }
    override func rightMouseDown(with e: NSEvent) { button(e, GHOSTTY_MOUSE_PRESS, GHOSTTY_MOUSE_RIGHT) }
    override func rightMouseUp(with e: NSEvent) { button(e, GHOSTTY_MOUSE_RELEASE, GHOSTTY_MOUSE_RIGHT) }
    override func otherMouseDown(with e: NSEvent) { button(e, GHOSTTY_MOUSE_PRESS, GHOSTTY_MOUSE_MIDDLE) }
    override func otherMouseUp(with e: NSEvent) { button(e, GHOSTTY_MOUSE_RELEASE, GHOSTTY_MOUSE_MIDDLE) }
    override func mouseMoved(with e: NSEvent) { position(e) }
    override func mouseDragged(with e: NSEvent) { position(e) }
    override func rightMouseDragged(with e: NSEvent) { position(e) }

    override func updateTrackingAreas() {
        trackingAreas.forEach(removeTrackingArea)
        addTrackingArea(NSTrackingArea(rect: bounds, options: [.mouseMoved, .activeInKeyWindow, .inVisibleRect],
                                       owner: self, userInfo: nil))
        super.updateTrackingAreas()
    }

    private func button(_ e: NSEvent, _ state: ghostty_input_mouse_state_e, _ b: ghostty_input_mouse_button_e) {
        guard let surface else { return }
        position(e)
        _ = ghostty_surface_mouse_button(surface, state, b, Self.mods(e.modifierFlags))
    }

    private func position(_ e: NSEvent) {
        guard let surface else { return }
        let p = convert(e.locationInWindow, from: nil)
        ghostty_surface_mouse_pos(surface, p.x, bounds.height - p.y, Self.mods(e.modifierFlags))
    }

    override func scrollWheel(with e: NSEvent) {
        guard let surface else { return }
        var x = e.scrollingDeltaX, y = e.scrollingDeltaY
        if e.hasPreciseScrollingDeltas { x *= 2; y *= 2 }
        // Bit 0 of scroll mods = precision (trackpad) scrolling.
        ghostty_surface_mouse_scroll(surface, x, y, e.hasPreciseScrollingDeltas ? 1 : 0)
    }

    // MARK: Clipboard (called from GhosttyRuntime's C callbacks)

    static func readClipboard(_ userdata: UnsafeMutableRawPointer?, state: UnsafeMutableRawPointer,
                              mimes: UnsafePointer<UnsafePointer<CChar>>, count: Int) -> ghostty_clipboard_read_result_e {
        guard let userdata else { return GHOSTTY_CLIPBOARD_READ_UNSUPPORTED }
        let view = Unmanaged<TerminalView>.fromOpaque(userdata).takeUnretainedValue()
        guard let surface = view.surface else { return GHOSTTY_CLIPBOARD_READ_UNSUPPORTED }
        let wantsText = (0..<count).contains { String(cString: mimes[$0]) == "text/plain" }
        guard wantsText, let text = NSPasteboard.general.string(forType: .string) else {
            return GHOSTTY_CLIPBOARD_READ_UNAVAILABLE
        }
        complete(surface, state: state, text: text, confirmed: false)
        return GHOSTTY_CLIPBOARD_READ_STARTED
    }

    /// Ghostty asks this before an unsafe paste (e.g. contains newlines).
    /// Keep its paste protection: ask the user.
    static func confirmReadClipboard(_ userdata: UnsafeMutableRawPointer?,
                                     confirm: UnsafePointer<ghostty_clipboard_confirm_s>,
                                     state: UnsafeMutableRawPointer) {
        guard let userdata else { return }
        let view = Unmanaged<TerminalView>.fromOpaque(userdata).takeUnretainedValue()
        guard let surface = view.surface else { return }
        let c = confirm.pointee
        var text: String?
        if let contents = c.contents {
            for i in 0..<c.contents_len where String(cString: contents[i].mime) == "text/plain" {
                text = String(decoding: Data(bytes: contents[i].data, count: contents[i].len), as: UTF8.self)
            }
        }
        guard let text else { ghostty_surface_deny_clipboard_request(surface, state); return }
        DispatchQueue.main.async {
            let alert = NSAlert()
            alert.messageText = "Paste into terminal?"
            alert.informativeText = String(text.prefix(500))
            alert.addButton(withTitle: "Paste")
            alert.addButton(withTitle: "Cancel")
            if alert.runModal() == .alertFirstButtonReturn {
                complete(surface, state: state, text: text, confirmed: true)
            } else {
                ghostty_surface_deny_clipboard_request(surface, state)
            }
        }
    }

    private static func complete(_ surface: ghostty_surface_t, state: UnsafeMutableRawPointer, text: String, confirmed: Bool) {
        "text/plain".withCString { mime in
            text.withCString { data in
                var content = ghostty_clipboard_content_s(mime: mime, data: data, len: strlen(data))
                withUnsafePointer(to: &content) { contentPtr in
                    var done = ghostty_clipboard_complete_s(
                        contents: contentPtr, contents_len: 1,
                        available: nil, available_len: 0,
                        confirmed: confirmed, remember: false)
                    ghostty_surface_complete_clipboard_request(surface, &done, state)
                }
            }
        }
    }
}
```

```swift
import AppKit
import DiffaryCore
import SwiftUI

@MainActor
final class AppModel: ObservableObject {
    let launch: Launch
    @Published private(set) var settings = Settings()
    @Published private(set) var theme = Theme.dark
    @Published private(set) var warnings: [String] = []
    @Published private(set) var changes: ChangeSet?
    @Published private(set) var gitError: String?
    @Published private(set) var selected: FileChange?
    @Published private(set) var diff: FileDiff?

    private var repoWatcher: FileWatcher?
    private var configWatcher: FileWatcher?
    private var appearanceObserver: NSKeyValueObservation?
    private let work = DispatchQueue(label: "diffary.git")

    var themesDir: URL { launch.configFile.deletingLastPathComponent().appendingPathComponent("themes") }

    init(launch: Launch) {
        self.launch = launch
        reloadSettings()
        refresh()
        repoWatcher = FileWatcher(paths: [launch.repo], ignoring: ["/.git/objects/", "/.git/logs/"]) { [weak self] in
            DispatchQueue.main.async { self?.refresh() }
        }
        repoWatcher?.start()
        // Watch the config dir, not the file: editors save by replacing it.
        let configDir = launch.configFile.deletingLastPathComponent()
        if FileManager.default.fileExists(atPath: configDir.path) {
            configWatcher = FileWatcher(paths: [configDir], debounce: 0.1) { [weak self] in
                DispatchQueue.main.async { self?.reloadSettings() }
            }
            configWatcher?.start()
        }
        appearanceObserver = NSApp.observe(\.effectiveAppearance) { [weak self] _, _ in
            DispatchQueue.main.async { self?.reloadSettings() }
        }
    }

    func reloadSettings() {
        let load = SettingsLoader.load(file: launch.configFile, overrides: launch.overrides)
        let isDark = NSApp.effectiveAppearance.bestMatch(from: [.darkAqua, .aqua]) == .darkAqua
        let resolved = ThemeLoader.resolve(name: load.settings.theme, systemIsDark: isDark, themesDir: themesDir)
        let baseChanged = load.settings.baseBranch != settings.baseBranch
        let diffChanged = load.settings.diffOptions != settings.diffOptions
        settings = load.settings
        theme = resolved.theme
        warnings = load.warnings + resolved.warnings
        if baseChanged { refresh() } else if diffChanged { loadDiff() }
    }

    func refresh() {
        let repo = launch.repo, base = settings.baseBranch
        work.async {
            let result = Result { try GitChanges(repo: repo, baseBranch: base).load() }
            DispatchQueue.main.async {
                switch result {
                case .success(let set):
                    self.changes = set
                    self.gitError = nil
                    // Keep the selection if the file is still changed; reload its diff.
                    if let sel = self.selected {
                        self.selected = set.files.first { $0.path == sel.path }
                        self.loadDiff()
                    }
                case .failure(let error):
                    self.changes = nil
                    self.gitError = "\(error)"
                }
            }
        }
    }

    func select(_ file: FileChange?) {
        selected = file
        loadDiff()
    }

    private func loadDiff() {
        guard let file = selected, let base = changes?.base else { diff = nil; return }
        let repo = launch.repo, options = settings.diffOptions
        work.async {
            let result = Result { try DiffLoader(repo: repo).load(file, base: base, options: options) }
            DispatchQueue.main.async {
                guard self.selected == file else { return }  // selection moved on
                switch result {
                case .success(let d): self.diff = d
                case .failure(let e): self.diff = nil; self.gitError = "\(e)"
                }
            }
        }
    }
}
```

```swift
struct LayoutView: View {

    var body: some View {
        let s = model.settings
        GeometryReader { geo in
            let horizontal = s.sidebar != .bottom
            let total = horizontal ? geo.size.width : geo.size.height
            let side = total * s.sidebarSize
            let main = mainArea.frame(width: horizontal ? total - side : nil,
                                      height: horizontal ? nil : total - side)
            let bar = sidebar.frame(width: horizontal ? side : nil, height: horizontal ? nil : side)
            switch s.sidebar {
            case .right: HStack(spacing: 0) { main; Divider(); bar }
```

```swift
import AppKit
import DiffaryCore

struct DiffView: View {
    @ObservedObject var model: AppModel

    var body: some View {
        let t = model.theme
        Group {
            switch model.diff {
            case nil: placeholder(model.selected == nil ? "Select a file" : "Loading…")
            case .binary?: placeholder("Binary file")
            case .tooLarge(let n)?: placeholder("Diff too large (\(n) lines). Open it in the terminal.")
            case .empty?: placeholder("No differences with current options")
            case .text(let hunks)?:
                ScrollView([.vertical, .horizontal]) {
                    LazyVStack(alignment: .leading, spacing: 0) {
                        ForEach(Array(hunks.enumerated()), id: \.offset) { _, hunk in
                            Text(hunk.header).foregroundStyle(Color(t.hunkHeader)).padding(.vertical, 2)
                            if model.settings.diffView == .unified {
                                ForEach(Array(hunk.lines.enumerated()), id: \.offset) { _, line in unifiedRow(line) }
                            } else {
                                ForEach(Array(hunk.sideBySideRows.enumerated()), id: \.offset) { _, row in
                                    HStack(spacing: 0) {
                                        half(row.left, number: row.left?.oldNumber)
                                        Divider()
                                        half(row.right, number: row.right?.newNumber)
                                    }
                                }
                            }
                        }
                    }
                    .padding(6)
                }
            }
        }
        .font(.custom(model.settings.font, size: model.settings.fontSize))
        .foregroundStyle(Color(t.foreground))
        .background(Color(t.background))
    }

    private func placeholder(_ s: String) -> some View {
        Text(s).foregroundStyle(Color(model.theme.lineNumber)).frame(maxWidth: .infinity, maxHeight: .infinity)
    }

    private func unifiedRow(_ line: DiffLine) -> some View {
        HStack(spacing: 8) {
            num(line.oldNumber); num(line.newNumber)
            Text(marker(line.kind)).foregroundStyle(Color(fg(line.kind)))
            code(line.text)
        }
        .frame(maxWidth: .infinity, alignment: .leading)
        .background(Color(bg(line.kind)))
    }

    private func half(_ line: DiffLine?, number: Int?) -> some View {
        HStack(spacing: 8) {
            num(number)
            if let line { code(line.text) } else { Text("") }
        }
        .frame(minWidth: 300, maxWidth: .infinity, alignment: .leading)
        .background(Color(line.map { bg($0.kind) } ?? RGBA(r: 0, g: 0, b: 0, a: 0)))
    }

    private func num(_ n: Int?) -> some View {
        Text(n.map(String.init) ?? "").foregroundStyle(Color(model.theme.lineNumber))
            .frame(width: 40, alignment: .trailing)
    }

    private func code(_ text: String) -> Text {
        guard model.settings.syntaxHighlight,
              let language = SyntaxHighlighter.language(for: model.selected?.path),
              let attr = SyntaxHighlighter.shared.highlight(text, language: language, theme: model.theme)
              // \.appKit scope keeps Highlightr's NSColor foregrounds visible to SwiftUI.
              let styled = try? AttributedString(attr, including: \.appKit)
        else { return Text(text) }
        return Text(styled)
    }

    private func marker(_ k: DiffLineKind) -> String { k == .added ? "+" : k == .removed ? "-" : " " }
    private func fg(_ k: DiffLineKind) -> RGBA {
        k == .added ? model.theme.addedFg : k == .removed ? model.theme.removedFg : model.theme.lineNumber
    }
    private func bg(_ k: DiffLineKind) -> RGBA {
        k == .added ? model.theme.addedBg : k == .removed ? model.theme.removedBg : RGBA(r: 0, g: 0, b: 0, a: 0)
    }

@MainActor
final class SyntaxHighlighter {
    static let shared = SyntaxHighlighter()
    private let highlightr = Highlightr()
    private var themeKey: String?

    static func language(for path: String?) -> String? {
        guard let ext = path.map({ ($0 as NSString).pathExtension.lowercased() }), !ext.isEmpty else { return nil }
        let map = ["swift": "swift", "ts": "typescript", "tsx": "typescript", "js": "javascript", "jsx": "javascript",
                   "py": "python", "rb": "ruby", "go": "go", "rs": "rust", "zig": "zig", "c": "c", "h": "c",
                   "cpp": "cpp", "m": "objectivec", "java": "java", "kt": "kotlin", "sh": "bash", "zsh": "bash",
                   "json": "json", "yml": "yaml", "yaml": "yaml", "toml": "ini", "md": "markdown",
                   "html": "xml", "css": "css", "sql": "sql"]
        return map[ext]
    }

    func highlight(_ code: String, language: String, theme: Theme) -> NSAttributedString? {
        guard let highlightr else { return nil }
        let key = theme.syntax.map { "\($0.key)\($0.value.hex)" }.sorted().joined() + theme.foreground.hex
        if key != themeKey {
            themeKey = key
            func rgb(_ c: RGBA?) -> String { String((c ?? theme.foreground).hex.prefix(7)) }
            let css = """
            .hljs{color:\(rgb(theme.foreground));background:transparent}
            .hljs-keyword,.hljs-built_in,.hljs-literal{color:\(rgb(theme.syntax["keyword"]))}
            .hljs-string,.hljs-regexp{color:\(rgb(theme.syntax["string"]))}
            .hljs-comment,.hljs-quote{color:\(rgb(theme.syntax["comment"]))}
            .hljs-number{color:\(rgb(theme.syntax["number"]))}
            .hljs-type,.hljs-title,.hljs-class{color:\(rgb(theme.syntax["type"]))}
            """
            highlightr.theme = Highlightr.Theme(themeString: css)
        }
        return highlightr.highlight(code, as: language, fastRender: true)
    }
```

````markdown
# diffary

A Ghostty terminal with a live sidebar of what your branch changed vs its base.

## Install

```sh
scripts/bundle.sh --install                       # → ~/Applications/Diffary.app
ln -sf "$PWD/scripts/diffary" ~/.local/bin/diffary
```

Needs the Command Line Tools (no Xcode) and `gh` access to this repo's releases (GhosttyKit is built in CI).

## Use

```sh
diffary [repo] [--config FILE] [--base BRANCH] [--set key=value]...
```

## Config

`$XDG_CONFIG_HOME/diffary/config.toml` (default `~/.config/diffary/config.toml`). Reloaded on save.
See `docs/superpowers/specs/2026-11-04-diffary-design.md` for every key and the theme format.
Themes: `~/.config/diffary/themes/<name>.toml`, selected with `[appearance] theme = "<name>"`.
Terminal settings come from Ghostty's own config.

## Bumping Ghostty

Put a new sha in `GHOSTTY_COMMIT` and push. CI publishes a new GhosttyKit; `scripts/fetch-ghosttykit.sh` picks it up.
The embedding API isn't stable, so expect to fix `TerminalView`/`GhosttyRuntime` after a bump.
````
